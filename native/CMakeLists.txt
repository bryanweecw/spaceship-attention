cmake_minimum_required(VERSION 3.20)
project(spaceship_attention LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/llama.cpp llama)
add_executable(attention-runner runner.cpp)
target_link_libraries(attention-runner PRIVATE llama)
target_include_directories(attention-runner PRIVATE ../vendor/llama.cpp/vendor)
